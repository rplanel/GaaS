## Mock GalaxyFetch Service

Blendtype's `GalaxyFetch` is a `Context.Tag`, making it trivial to replace with mocks:

### Success Layer

```typescript [test/fixtures.ts]
import { GalaxyFetch } from 'blendtype'
import { Effect, Layer } from 'effect'

function createSuccessLayer<T>(response: T): Layer.Layer<GalaxyFetch> {
  return Layer.succeed(
    GalaxyFetch,
    (async () => response) as unknown as ReturnType<typeof $fetch.create>
  )
}
```

### Failure Layer

```typescript
function createFailureLayer(error: Error): Layer.Layer<GalaxyFetch> {
  return Layer.succeed(
    GalaxyFetch,
    (async () => { throw error }) as unknown as ReturnType<typeof $fetch.create>
  )
}
```

## Testing Effect Operations

### Success Case

```typescript [test/workflows.test.ts]
import { getWorkflowEffect } from 'blendtype'
import { Effect } from 'effect'
import { describe, expect, it } from 'vitest'

describe('workflows', () => {
  it('should fetch workflow by ID', () =>
    Effect.gen(function* () {
      const workflow = yield* getWorkflowEffect('test-workflow-id')
      expect(workflow).toEqual(mockGalaxyWorkflow)
    }).pipe(
      Effect.provide(createSuccessLayer(mockGalaxyWorkflow)),
      Effect.runPromise
    ))
})
```

### Error Case with Effect.exit

```typescript
import { Effect } from 'effect'

it('should handle network failures', () =>
  Effect.gen(function* () {
    const exit = yield* Effect.exit(
      getWorkflowEffect('test-id')
    )

    if (Exit.isFailure(exit)) {
      const error = exit.cause
      expect(error).toBeInstanceOf(WorkflowError)
      expect(error.message).toContain('Error getting workflow test-id')
    }
  }).pipe(
    Effect.provide(createFailureLayer(new Error('Connection refused'))),
    Effect.runPromise
  ))
```

## Testing Promise Wrappers

```typescript
import { getWorkflow } from 'blendtype'

it('should resolve with workflow data', async () => {
  const result = await getWorkflow(
    'test-id',
    createSuccessLayer(mockGalaxyWorkflow)
  )
  expect(result).toEqual(mockGalaxyWorkflow)
})
```

## URL Tracking

Verify the correct API endpoint is called:

```typescript
function createUrlTrackingLayer<T>(response: T) {
  let lastUrl = ''

  const fetch = (async (url: string) => {
    lastUrl = url
    return response
  }) as unknown as ReturnType<typeof $fetch.create>

  return {
    layer: Layer.succeed(GalaxyFetch, fetch),
    getLastUrl: () => lastUrl
  }
}

// Usage
const { layer, getLastUrl } = createUrlTrackingLayer(mockGalaxyWorkflow)

await Effect.runPromise(
  Effect.provide(getWorkflowEffect('test-id'), layer)
)

expect(getLastUrl()).toBe('api/workflows/test-id')
```

## HTTP Status Mocking

Test specific HTTP error codes:

```typescript
function createHttpErrorLayer(statusCode: number, statusText: string) {
  return Layer.succeed(
    GalaxyFetch,
    (async () => {
      const error = new Error(statusText)
      ;(error as any).statusCode = statusCode
      throw error
    }) as unknown as ReturnType<typeof $fetch.create>
  )
}

// Test 404 Not Found
Effect.provide(operation, createHttpErrorLayer(404, 'Not Found'))

// Test 503 Service Unavailable
function createServiceUnavailableLayer() {
  return createHttpErrorLayer(503, 'Service Unavailable')
}
```

## Running Tests

```bash [Terminal]
# Run all tests with coverage
pnpm test

# Run specific test file
pnpm vitest run test/workflows.test.ts

# Watch mode
pnpm vitest watch
```

::tip
Always test the Effect version directly and the Promise wrapper indirectly. The Promise wrapper delegates to the Effect version, so testing the Effect version covers the core logic.
::
